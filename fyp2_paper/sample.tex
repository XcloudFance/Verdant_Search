\section{Fulfilment of Project Objectives}
%==================================================================

\begin{frame}{Project Objectives \#1}
  \begin{columns}[t]
    \column{.50\textwidth}
    \begin{block}{Objective}
      To \textbf{identify} existing search engine architectures,
      learning-to-rank approaches, vision language models, and
      retrieval-augmented generation techniques through literature review.
    \end{block}
    \vspace{.4em}
    \begin{block}{Fulfilment Methods}
      \begin{itemize}
        \item Literature Review conducted
          \begin{itemize}\scriptsize
            \item 9 research areas surveyed (architecture, keyword search,
              vector embeddings, ANN, classical LTR, LLM ranking, VLMs,
              RAG systems, benchmark datasets)
            \item System Archeology for existing applications
          \end{itemize}
        \item Comparison with existing systems (Google, Perplexity,
          MMDocIR, RankZephyr)
      \end{itemize}
    \end{block}
    \column{.48\textwidth}
    \begin{alertblock}{Research Gaps Identified}
      \begin{enumerate}\small
        \item \textbf{No multimodal reranking} — all existing LTR methods text-only
        \item \textbf{Visual evidence underutilised} in RAG systems
        \item \textbf{One-shot retrieval} — no iterative LLM-driven refinement
        \item \textbf{VLM inference cost} — efficiency challenges at scale
      \end{enumerate}
    \end{alertblock}
    \vspace{.4em}
    \begin{exampleblock}{Project Outcome}
      \small FR and NFR of Verdant-Search address all identified gaps.
      Evaluation results confirm this.
    \end{exampleblock}
  \end{columns}
\end{frame}

\begin{frame}{Project Objectives \#2}
  \begin{columns}[t]
    \column{.48\textwidth}
    \begin{block}{Objective}
      To \textbf{design and develop} a multimodal search engine
      integrating distributed crawling, hybrid retrieval, multimodal
      listwise reranking, and iterative RAG with query rewriting.
    \end{block}
    \vspace{.4em}

\begin{frame}{Health Data Fetching — Incremental Index Update Logic}
  \vspace{-4.0em}
  \begin{columns}[t]
    \column{.42\textwidth}
    \begin{block}{Difficulty}
      Ensuring \textbf{no documents are missed} during incremental index
      updates while \textbf{avoiding re-indexing} of unchanged content.
    \end{block}
    \vspace{.3em}
    \begin{block}{Solution}
      Tracks the \textbf{last indexed document's timestamp and content hash}
      per source, using them as the starting checkpoint for the next
      crawl cycle.
    \end{block}
    \vspace{.3em}
    \begin{block}{Logic Summary}
      \scriptsize
      \begin{itemize}
        \item Has checkpoint $\rightarrow$ \texttt{startTime} = last indexed timestamp
        \item No checkpoint $\rightarrow$ \texttt{startTime} = full historical crawl
        \item Gap $>$ 7 days $\rightarrow$ batch crawl in weekly windows
        \item Gap $\leq$ 7 days $\rightarrow$ incremental crawl to \texttt{currentTime}
      \end{itemize}
    \end{block}
    \column{.56\textwidth}
    \vspace{3.5em}

\begin{frame}{Notification Cron Job — Search Quality Monitoring}
  \vspace{-0.8em}
  \begin{columns}[t]
    \column{.42\textwidth}
    \begin{block}{Difficulty}
      Ensuring the \textbf{same quality alert is not triggered multiple times}
      within a monitoring window.
    \end{block}
    \vspace{.3em}
    \begin{block}{Solution}
      Tracks \textbf{timestamps of previously sent alerts} per metric type
      and compares against current time and threshold before sending.
    \end{block}

      \begin{enumerate}
        \item P90 latency $>$ 4s $\rightarrow$ alert if last same alert $>$ 1 hr ago
        \item Index staleness $>$ 24 hrs $\rightarrow$ alert if last same alert $>$ 6 hrs ago
        \item Reranker error rate $>$ 5\% $\rightarrow$ alert if last same alert $>$ 30 min ago
      \end{enumerate}
    \end{block}
