\section{Module 2: Search and Retrieval}
%------------------------------------------------

\begin{frame}{Module 2: Search and Retrieval - Implementation Status}
    \large
    \textbf{Implemented Features:}
    \begin{itemize}
        \item \checkmark \textbf{Text Input \& Query Processing}
        \begin{itemize}
            \normalsize
            \item User query submission through search interface
            \item Query sanitization and normalization
            \item Query expansion with synonyms
        \end{itemize}
        
        \vspace{0.5em}
        \item \checkmark \textbf{Keyword Search (BM25)}
        \begin{itemize}
            \normalsize
            \item Inverted index with term posting lists
            \item BM25 probabilistic ranking
            \item Fast lexical matching
        \end{itemize}
        
        \vspace{0.5em}
        \item \checkmark \textbf{Vector Search (HNSW)}
        \begin{itemize}
            \normalsize
            \item Sentence-BERT query encoding
            \item HNSW approximate nearest neighbor search
            \item Semantic similarity matching
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Module 2: Search and Retrieval - Implementation Status (Cont.)}
    \large
    \textbf{Implemented Features (Continued):}
    \begin{itemize}
        \item \checkmark \textbf{Hybrid Search Fusion}
        \begin{itemize}
            \normalsize
            \item Reciprocal Rank Fusion (RRF) combining BM25 and vector scores
            \item Weighted score combination
            \item Metadata-based pre-filtering
        \end{itemize}
        
        \vspace{0.5em}
        \item \checkmark \textbf{Search Results Display}
        \begin{itemize}
            \normalsize
            \item Ranked document list with relevance scores
            \item Document snippets with highlighted keywords
            \item Metadata display (filename, page number, timestamp)
        \end{itemize}

\begin{frame}{Module 5: AI Iterative Chat (RAG) - Implementation Status}
    \large
    \textbf{Implemented Features:}
    \begin{itemize}
        \item \checkmark \textbf{Query Processing \& Validation}
        \begin{itemize}
            \normalsize

            \item Query length and format checking
        \end{itemize}
        
        \vspace{0.5em}
        \item \checkmark \textbf{Context Injection}
        \begin{itemize}
            \normalsize
            \item Retrieved context injection into LLM prompt
            \item Context formatting with document metadata
            \item Structured prompt engineering
        \end{itemize}
        
        \vspace{0.5em}
        \item \checkmark \textbf{AI Answer Generation}
        \begin{itemize}
            \normalsize
            \item LLM-based answer generation grounded in context
            \item Hallucination prevention (context-only responses)
            \item Markdown-formatted answer display
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Module 5: AI Iterative Chat (RAG) - Implementation Status (Cont.)}
    \large
    \textbf{Implemented Features (Continued):}
    \begin{itemize}
        \item \checkmark \textbf{Citation \& Source References}
        \begin{itemize}
            \normalsize
            \item Inline citations with source document links
            \item Citations include page numbers and timestamps
            \item Highlighted cited text in source documents
            \item Formatted source references (title, author, timestamp)
        \end{itemize}
        
        \vspace{0.5em}
        \item \checkmark \textbf{Real-time Streaming}
        \begin{itemize}
            \normalsize
            \item Server-sent events (SSE) for streaming responses
            \item Progressive answer display
            \item Low latency user experience
        \end{itemize}
        
        \vspace{0.5em}
        \item \checkmark \textbf{Knowledge Base Awareness}
        \begin{itemize}

            \item Clear indication of knowledge base limitations
            \item Entailment checking for answer grounding validation

    \end{itemize}
\end{frame}
